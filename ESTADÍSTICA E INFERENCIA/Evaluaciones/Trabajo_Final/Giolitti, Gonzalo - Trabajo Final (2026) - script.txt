# =============================================================================
# 2. SITUACIÓN PROBLEMÁTICA 1: CONTRASTE DE HIPÓTESIS
#    Variable en estudio: presión arterial sistólica (mmHg)
#    Unidad elemental: paciente
#    Factor: presencia de enfermedad cardiovascular; niveles: sin enfermedad (0) y con enfermedad (1)
#    H0) mu_con = mu_sin     H1) mu_con > mu_sin     alfa = 0,05
# =============================================================================
sink(file.path(dir_salidas, "salida_2_contraste.txt"), split = TRUE)
cat("=== 2. Contraste de hipótesis: presión sistólica según enfermedad cardiovascular ===\n")
alfa <- 0.05
sis_sin <- datos$ap_hi[datos$cardio == 0]
sis_con <- datos$ap_hi[datos$cardio == 1]

cat("\n--- Media, mediana y desvío por grupo ---\n")
print(sapply(split(datos$ap_hi, datos$cardio_f),
             function(x) c(n = length(x), media = mean(x), mediana = median(x),
                           desvio = sd(x), Q1 = unname(quantile(x, 0.25)), Q3 = unname(quantile(x, 0.75)))))

# Supuestos. shapiro.test admite como máximo 5000 datos: muestra aleatoria de 5000 por grupo
set.seed(2026)
muestras_h1 <- lapply(split(datos$ap_hi, datos$cardio_f), sample, size = 5000)
cat("\n--- Normalidad: Shapiro-Wilk por grupo (muestra aleatoria de 5000) ---\n")
print(lapply(muestras_h1, shapiro.test))
cat("\n--- Homocedasticidad: test de Bartlett ---\n")
print(bartlett.test(list(sis_sin, sis_con)))

# No hay normalidad ni homocedasticidad -> test de Wilcoxon (suma de rangos), muestras independientes.
# alternative = "greater": la sistólica del grupo con enfermedad (x) es mayor que la del grupo sin enfermedad (y)
cat("\n--- Test de Wilcoxon (alternativa a la derecha) ---\n")
w1 <- wilcox.test(sis_con, sis_sin, paired = FALSE, alternative = "greater", conf.level = 0.95)
print(w1)
cat("\n--- Diferencias entre grupos ---\n")
print(c(dif_medias = mean(sis_con) - mean(sis_sin), dif_medianas = median(sis_con) - median(sis_sin)))
sink()

sh1 <- lapply(muestras_h1, shapiro.test); bt1 <- bartlett.test(list(sis_sin, sis_con))
guardar_valor("h1_media_sin", mean(sis_sin)); guardar_valor("h1_media_con", mean(sis_con))
guardar_valor("h1_med_sin", median(sis_sin)); guardar_valor("h1_med_con", median(sis_con))
guardar_valor("h1_sd_sin", sd(sis_sin)); guardar_valor("h1_sd_con", sd(sis_con))
guardar_valor("h1_q1_sin", quantile(sis_sin, 0.25)); guardar_valor("h1_q3_sin", quantile(sis_sin, 0.75))
guardar_valor("h1_q1_con", quantile(sis_con, 0.25)); guardar_valor("h1_q3_con", quantile(sis_con, 0.75))
guardar_valor("h1_W_sin", sh1[[1]]$statistic); guardar_valor("h1_W_con", sh1[[2]]$statistic)
guardar_valor("h1_sh_p_sin", sh1[[1]]$p.value); guardar_valor("h1_sh_p_con", sh1[[2]]$p.value)
guardar_valor("h1_bart_K", bt1$statistic); guardar_valor("h1_bart_gl", bt1$parameter); guardar_valor("h1_bart_p", bt1$p.value)
guardar_valor("h1_wilcox_W", w1$statistic); guardar_valor("h1_wilcox_p", w1$p.value)
guardar_valor("h1_dif_medias", mean(sis_con) - mean(sis_sin)); guardar_valor("h1_dif_medianas", median(sis_con) - median(sis_sin))

# =============================================================================
# 3. SITUACIÓN PROBLEMÁTICA 2: CLASIFICACIÓN CON ÁRBOLES DE DECISIÓN
#    Procedimiento de la Unidad 3, clase 7: variable objetivo como factor, partición 70/30
#    (sample_frac y setdiff), rpart, rpart.plot, predict sobre la prueba y matriz de confusión (caret).
#    Variable objetivo: cardio_f (sin enfermedad / con enfermedad).
#    Predictores: edad_anios, gender, height, weight, ap_hi, ap_lo, cholesterol, gluc, smoke, alco, active.
#    No se usan id, age (es la edad en días, duplicaría edad_anios) ni cardio en otra forma.
# =============================================================================
sink(file.path(dir_salidas, "salida_3_arboles.txt"), split = TRUE)
cat("=== 3. Clasificación: árboles de decisión ===\n")

set.seed(2026)
datos_entrenamiento <- sample_frac(datos, .7)            # 70% para entrenar
datos_prueba <- setdiff(datos, datos_entrenamiento)      # el 30% restante para probar
cat("\nPacientes en entrenamiento:", nrow(datos_entrenamiento), " Pacientes en prueba:", nrow(datos_prueba), "\n")
cat("Pacientes (id) que están en ambos conjuntos:", length(intersect(datos_entrenamiento$id, datos_prueba$id)), "\n")
cat("\nClases en la prueba:\n"); print(table(datos_prueba$cardio_f))
exactitud_base <- max(table(datos_prueba$cardio_f)) / nrow(datos_prueba)
cat("Exactitud de predecir siempre la clase más frecuente:", exactitud_base, "\n")

formula_arbol <- cardio_f ~ edad_anios + gender_f + height + weight + ap_hi + ap_lo +
  cholesterol_f + gluc_f + smoke_f + alco_f + active_f

# Cuatro árboles con los mismos datos de entrenamiento, de menor a mayor complejidad.
# cp (parámetro de complejidad) más chico = el árbol se divide más veces.
arbol_defecto <- rpart(formula = formula_arbol, data = datos_entrenamiento)      # cp = 0,01 (por defecto)
arbol_mediano <- rpart(formula = formula_arbol, data = datos_entrenamiento,
                       control = rpart.control(cp = 0.001))
arbol_grande  <- rpart(formula = formula_arbol, data = datos_entrenamiento,
                       control = rpart.control(cp = 0.0005))
arbol_completo <- rpart(formula = formula_arbol, data = datos_entrenamiento,
                        control = rpart.control(cp = 0))                         # sin poda
arboles <- list("Árbol por defecto (cp = 0,01)" = arbol_defecto,
                "Árbol mediano (cp = 0,001)" = arbol_mediano,
                "Árbol grande (cp = 0,0005)" = arbol_grande,
                "Árbol completo (cp = 0)" = arbol_completo)

cat("\n--- Cantidad de hojas de cada árbol ---\n")
hojas <- sapply(arboles, function(a) sum(a$frame$var == "<leaf>"))
print(hojas)

# Predicción sobre los datos de prueba y matriz de confusión (la clase positiva es "Con enfermedad")
resultados <- data.frame()
matrices <- list()
for (i in seq_along(arboles)) {
  a <- arboles[[i]]
  pred_prueba <- predict(a, newdata = datos_prueba, type = "class")
  pred_entren <- predict(a, newdata = datos_entrenamiento, type = "class")
  cm <- confusionMatrix(pred_prueba, datos_prueba[["cardio_f"]], positive = "Con enfermedad")
  cm_e <- confusionMatrix(pred_entren, datos_entrenamiento[["cardio_f"]], positive = "Con enfermedad")
  matrices[[i]] <- cm
  cat("\n=======================================================================\n")
  cat(names(arboles)[i], "\n")
  print(cm)
  resultados <- rbind(resultados, data.frame(
    arbol = names(arboles)[i], hojas = unname(hojas[i]),
    exactitud_entrenamiento = unname(cm_e$overall["Accuracy"]),
    exactitud_prueba = unname(cm$overall["Accuracy"]), kappa_prueba = unname(cm$overall["Kappa"]),
    sensibilidad = unname(cm$byClass["Sensitivity"]), especificidad = unname(cm$byClass["Specificity"])))
}
cat("\n--- Comparación de los árboles ---\n")
print(resultados, row.names = FALSE, digits = 4)

# Árbol elegido: el más simple cuya exactitud en la prueba esté a menos de 0,001 de la mejor
# (la diferencia es mínima y el árbol chico se puede graficar e interpretar)
i_elegido <- which(resultados$exactitud_prueba >= max(resultados$exactitud_prueba) - 0.001)[1]
arbol_elegido <- arboles[[i_elegido]]
cat("\nÁrbol elegido:", names(arboles)[i_elegido], "\n")
cat("\n--- Árbol elegido (reglas) ---\n")
print(arbol_elegido)
cat("\n--- Primera división del árbol elegido ---\n")
print(arbol_elegido$splits[1, , drop = FALSE])
cat("\nImportancia de las variables en el árbol elegido:\n")
print(round(arbol_elegido$variable.importance, 1))
sink()

cm_el <- matrices[[i_elegido]]
guardar_valor("n_entrenamiento", nrow(datos_entrenamiento)); guardar_valor("n_prueba", nrow(datos_prueba))
guardar_valor("prueba_sin", sum(datos_prueba$cardio == 0)); guardar_valor("prueba_con", sum(datos_prueba$cardio == 1))
guardar_valor("exactitud_base", exactitud_base); guardar_valor("i_elegido", i_elegido)
for (i in seq_len(nrow(resultados))) {
  for (campo in c("hojas", "exactitud_entrenamiento", "exactitud_prueba", "kappa_prueba", "sensibilidad", "especificidad"))
    guardar_valor(paste0("arbol", i, "_", campo), resultados[i, campo])
}
guardar_valor("tn", cm_el$table[1, 1]); guardar_valor("fp", cm_el$table[2, 1])
guardar_valor("fn", cm_el$table[1, 2]); guardar_valor("tp", cm_el$table[2, 2])
